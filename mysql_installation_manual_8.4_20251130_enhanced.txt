>>>>>>> create drop procedure::::::::::

sqlp6
create database dba;

###create procedure###
delimiter //
create procedure dba.truncdrop(dbn varchar(1000),tbd varchar(1000),op varchar(1000),tt varchar(1000),pn varchar(1000))
al: begin
if (lower(dbn)='performance_schema' or lower(dbn)='mysql' or lower(dbn)='sys' ) then
select 'you cannot drop system tables';
leave al;
end if;
    case 
     when upper(op)='TRUNCATE' and upper(tt)='TABLE'
        THEN
        SET @truncatetable = CONCAT ('TRUNCATE TABLE ',dbn,'.', tbd);
        PREPARE trunctbl FROM @truncatetable;
        EXECUTE trunctbl ;
        show warnings;
        DEALLOCATE PREPARE trunctbl ;
    when upper(op)='DROP' and upper(tt)='VIEW'
        THEN
        SET @dropview = CONCAT ('DROP VIEW ',dbn,'.', tbd);
        PREPARE dropview_exe FROM @dropview;
        EXECUTE dropview_exe ;
        show warnings;
        DEALLOCATE PREPARE dropview_exe ;    
        
    when upper(op)='DROP' and upper(tt)='TABLE'
        THEN
        SET @droptable = CONCAT ('DROP TABLE IF EXISTS ',dbn,'.', tbd);
        PREPARE droptbl FROM @droptable;
        EXECUTE droptbl ;
        show warnings;
        DEALLOCATE PREPARE droptbl ;
    when upper(op)='TRUNCATE' and upper(tt)='PARTITION'
    THEN
        SET @truncpart = CONCAT ('alter table ',dbn,'.', tbd,' truncate partition ',pn);
        PREPARE truncppart FROM @truncpart;
        EXECUTE truncppart ;
        show warnings;
        DEALLOCATE PREPARE truncppart ;
    when upper(op)='DROP' and upper(tt)='PARTITION'
    THEN
        SET @droppart = CONCAT ('alter table ',dbn,'.', tbd,' drop partition ',pn);
        PREPARE dropppart FROM @droppart;
        EXECUTE dropppart ;
        show warnings;
        DEALLOCATE PREPARE dropppart ;
    else 
        select 'Usage is: First parameter as database name,Second parameter as table name or view name ,Third parameter as operation,Fourth parameter as object type that can be table or view or partition,Fifth parameter as partition name';
    END case ;
end;
//
delimiter ;
exit

>>>>>>> how to use this procedure? you can send following instruction to app team::::::

For non-partitioned tables:
call dba.truncdrop('DB_Name','Table_Name','truncate or drop','table','');
e.g.
call dba.truncdrop('testdb','test_table','truncate','table','');
call dba.truncdrop('testdb','test_table','drop','table','');

For partitioned tables:
call dba.truncdrop('DB_Name','Table_Name','truncate or drop','partition','partition_name');
e.g.
call dba.truncdrop('testdb','test_table','truncate','partition','p2');
call dba.truncdrop('testdb','test_table','drop','partition','p2');

For views:
call dba.truncdrop('DB_Name','View_name','drop','view','');
